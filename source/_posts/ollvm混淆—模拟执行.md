## 思路

1.先用idaapi脚本，得到真实块和虚假块

2.然后用unicorn模拟器得到执行路径

3.最后用idapython批量化去跳转和虚假块

### 脚本1

```python
import idaapi
import idc

func_start    = 0x00000223C7CDB370
process_path  = 0x00000223C7CDB375
fun_ret       = 0x00000223C7CDFFE2
offset        = 0x223C7CB1000-0x400000

true_blocks = set()
fake_blocks = set()

func = idaapi.get_func(func_start)
if func is None:
    print("获取函数失败！")
else:
    flowchart = idaapi.FlowChart(func)
    print("analyzing function...")

    for block in flowchart:
        start_ea = block.start_ea
        end_ea   = idc.prev_head(block.end_ea)

        if start_ea == process_path:
            fake_blocks.add((start_ea, end_ea))
            for pred in block.preds():
                true_blocks.add((pred.start_ea, idc.prev_head(pred.end_ea)))
            continue
        if start_ea == fun_ret:
            true_blocks.add((start_ea, end_ea))
            continue
        if start_ea == func_start:
            true_blocks.add((start_ea, end_ea))

    for block in flowchart:
        key = (block.start_ea, idc.prev_head(block.end_ea))
        if key not in true_blocks:
            fake_blocks.add(key)
print("----------------------------------------------------")
for t in sorted(true_blocks):
    print(f"({hex(t[0] - offset)},{hex(t[1] - offset)}),", end="")
for t in sorted(fake_blocks):
    print(f"({hex(t[0]-offset)},{hex(t[1]-offset)}),",end="")
print("----------------------------------------------------")
```
